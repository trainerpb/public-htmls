Job Sequencing with Deadlines

The Greedy Strategy: The objective is to maximize total profit when you are given a set of jobs, each taking 1 unit of time, but restricted by strict deadlines. The Greedy paradigm dictates that we must prioritize the highest paying jobs. By sorting jobs by profit in descending order, we guarantee maximum revenue. Crucially, when placing a high-value job, we always assign it to the latest possible free time slot before its deadline, leaving early time slots open for subsequent jobs.
Time Complexity: $O(N^2)$

Sorting takes $O(N \log N)$. In the worst case, searching backwards for an empty slot for each job takes $O(N^2)$. (Can be optimized to $O(N \log N)$ using Disjoint Set Union).

Space Complexity: $O(N)$

Requires an array to keep track of which time slots have been filled, sized to the maximum deadline present in the dataset.

class Job {
    char id;
    int deadline, profit;
    
    public Job(char id, int deadline, int profit) {
        this.id = id;
        this.deadline = deadline;
        this.profit = profit;
    }
}

public void printJobScheduling(ArrayList<Job> arr, int maxDeadline) {
    // 1. Sort jobs by profit in descending order (Greedy Choice)
    Collections.sort(arr, (a, b) -> b.profit - a.profit);
    
    boolean[] slotFilled = new boolean[maxDeadline];
    char[] result = new char[maxDeadline];
    int totalProfit = 0;

    // 2. Iterate through the sorted jobs
    for (int i = 0; i < arr.size(); i++) {
        
        // 3. Search backwards from the job's deadline
        // (Convert 1-based deadline to 0-based index)
        int startSlot = Math.min(maxDeadline - 1, arr.get(i).deadline - 1);
        
        for (int j = startSlot; j >= 0; j--) {
            // If the slot is empty, place the job!
            if (!slotFilled[j]) {
                slotFilled[j] = true;
                result[j] = arr.get(i).id;
                totalProfit += arr.get(i).profit;
                break; // Job assigned, break the inner loop
            }
        }
    }
    
    System.out.println("Max Profit: " + totalProfit);
}
def printJobScheduling(jobs, max_deadline):
    
    # 1. Sort jobs by profit in descending order (Greedy Choice)
    jobs.sort(key=lambda x: x['profit'], reverse=True)
    
    slot_filled = [False] * max_deadline
    result = ['-1'] * max_deadline
    total_profit = 0

    # 2. Iterate through the sorted jobs
    for i in range(len(jobs)):
        
        # 3. Search backwards from the job's deadline
        # (Convert 1-based deadline to 0-based index)
        start_slot = min(max_deadline - 1, jobs[i]['deadline'] - 1)
        
        for j in range(start_slot, -1, -1):
            # If the slot is empty, place the job!
            if not slot_filled[j]:
                slot_filled[j] = True
                result[j] = jobs[i]['id']
                total_profit += jobs[i]['profit']
                break # Job assigned, break the inner loop
                
    print(f"Max Profit: {total_profit}")
    return result
